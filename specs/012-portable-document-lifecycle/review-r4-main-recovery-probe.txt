"""Bounded local regression controls; no shared checkout writes."""
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
import pytest
from ai import test_portable_baseline as baseline

FROZEN = Path("/tmp/w003-r4-contract.aBwYIx/repo")
assert baseline.ROOT.resolve() == FROZEN.resolve()
CHILD = """
import importlib.util, json, pathlib, sys
source = pathlib.Path(sys.argv[1]).resolve()
spec = importlib.util.spec_from_file_location("classifier_control", source)
module = importlib.util.module_from_spec(spec)
exec(compile(source.read_bytes(), str(source), "exec", dont_inherit=True), module.__dict__)
assert pathlib.Path(module.__file__).resolve() == source
prefix = json.loads(sys.stdin.read())
row = {"reference":"src/api.py", "raw_reference":"src/api.py", "line":1,
       "source_kind":"inline_code", "prefix":prefix, "context":prefix}
print(json.dumps(module.classify_document_reference(row, {"src"})))
"""

@pytest.mark.parametrize("relative", [
    "scripts/amplai_docs.py",
    "tools/amplai-loop-kit/baseline/payload/scripts/amplai_docs.py",
])
@pytest.mark.parametrize("declaration", [True, False])
def test_classification_completes_within_local_process_bound(relative, declaration):
    prefix = "Requires both files " if declaration else "Dependencies: " + " " * 30 + "notes "
    try:
        result = subprocess.run(
            [sys.executable, "-I", "-S", "-B", "-c", CHILD, str(FROZEN / relative)],
            input=json.dumps(prefix), text=True, capture_output=True, timeout=3,
        )
    except subprocess.TimeoutExpired:
        pytest.fail("one short reference exceeded the local 3-second child-process bound")
    assert result.returncode == 0, result.stderr
    row = json.loads(result.stdout)
    assert row["required"] is declaration

@pytest.mark.parametrize("relative", ["scripts/amplai_docs.py", ".agents/skills/work/SKILL.md"])
def test_borrowed_baseline_reinstall_and_uninstall_preserve_mode(tmp_path, relative):
    target = tmp_path / relative
    target.parent.mkdir(parents=True)
    original = (baseline.BASELINE / "payload" / relative).read_bytes()
    target.write_bytes(original)
    target.chmod(0o600)
    for run in range(2):
        result = baseline.install_canary(tmp_path, "--bootstrap-baseline", "--no-git")
        assert result.returncode == 0, result.stderr
        assert target.stat().st_mode & 0o777 == 0o600, "borrowed mode changed on run %d" % run
        receipt = json.loads((tmp_path / ".ai-team/install/amplai-loop-kit.json").read_text())
        borrowed = receipt["baseline"]["files"][relative]
        assert borrowed["created"] is False
        assert borrowed["mode"] == 0o600
    result = baseline.install_canary(tmp_path, "--uninstall", "--no-git")
    assert result.returncode == 0, result.stderr
    assert target.read_bytes() == original
    assert target.stat().st_mode & 0o777 == 0o600
