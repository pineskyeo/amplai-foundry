"""Bounded native transport smoke. No product work, tools or live Store access."""
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parent
repo = root / "app"
repo.mkdir()
kit = Path("/Users/pinesky/workspace/amplai-foundry/tools/amplai-loop-kit")
env = {k: v for k, v in os.environ.items() if not k.startswith(("AMPLAI_", "PYTHON", "GIT_"))
       and k not in ("CLAUDECODE", "CLAUDE_PROJECT_DIR", "CLAUDE_ENV_FILE")}
env["PYTHONDONTWRITEBYTECODE"] = "1"
installed = subprocess.run([sys.executable, "-B", str(kit / "install.py"), "--target", str(repo),
    "--app-id", "native-canary", "--project-id", "disposable-canary", "--bootstrap-baseline"],
    cwd=repo, env=env, capture_output=True, timeout=30)
(root / "install.stdout").write_bytes(installed.stdout)
(root / "install.stderr").write_bytes(installed.stderr)
if installed.returncode:
    print(json.dumps({"stage": "install", "exit_code": installed.returncode}), flush=True)
    sys.exit(1)
subprocess.run(["git", "init", "--quiet"], cwd=repo, env=env, check=True, capture_output=True)
spec = importlib.util.spec_from_file_location("actual_installed_hosts", repo / "scripts/amplai_hosts.py")
hosts = importlib.util.module_from_spec(spec)
spec.loader.exec_module(hosts)
token = "AMPLAI_NATIVE_TRANSPORT_CANARY_OK"
prompt = ("This is a transport-only integration smoke, not a development task. "
          "Do not call tools, delegate, inspect files, change files, access a Project Store, "
          "or perform any action. Respond with exactly this single line: " + token)
def inventory():
    return {p.relative_to(repo).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in repo.rglob("*") if p.is_file() and ".git" not in p.relative_to(repo).parts}
before = inventory()
results = []
for host, args in [
    ("claude-code", ["--no-session-persistence", "--tools", "", "--max-budget-usd", "1"]),
    ("codex", ["--ephemeral", "--sandbox", "read-only", "--color", "never"]),
]:
    print(json.dumps({"host": host, "status": "RUNNING"}), flush=True)
    adapter = hosts.get_host_adapter({"type": host, "args": args})
    command = adapter.build_command(prompt)
    try:
        result = subprocess.run(command, cwd=repo, env=env, capture_output=True, timeout=45)
        output, error, code = result.stdout, result.stderr, result.returncode
        timed_out = False
    except subprocess.TimeoutExpired as exc:
        output, error, code = exc.stdout or b"", exc.stderr or b"", None
        timed_out = True
    out = root / (host + ".stdout")
    err = root / (host + ".stderr")
    out.write_bytes(output)
    err.write_bytes(error)
    events = []
    try:
        events = [json.loads(output.decode())] if host == "claude-code" else [
            json.loads(line) for line in output.decode().splitlines() if line.strip()]
    except (ValueError, UnicodeError):
        pass
    texts = []
    for event in events:
        if host == "claude-code":
            texts.append(event.get("result", ""))
        elif event.get("type") == "item.completed":
            item = event.get("item", {})
            if item.get("type") == "agent_message":
                texts.append(item.get("text", ""))
    match = any(value.strip() == token for value in texts)
    summary = {"host": host, "exit_code": code, "timed_out": timed_out,
               "answer_exact": match, "transport_pass": code == 0 and match,
               "continuation_id_observed": bool(adapter.continuation_id_from_output(str(out))),
               "stdout_sha256": hashlib.sha256(output).hexdigest(), "stdout_bytes": len(output),
               "stderr_sha256": hashlib.sha256(error).hexdigest(), "stderr_bytes": len(error),
               "work_entry": adapter.public_work_entry, "design_entry": adapter.public_design_entry}
    results.append(summary)
    print(json.dumps(summary), flush=True)
after = inventory()
drift = sorted(k for k in set(before) | set(after) if before.get(k) != after.get(k))
summary = {"schema_version": 1, "results": results, "repository_content_drift": drift,
           "scope": "Actual installed adapters and native CLI response transport only",
           "limits": ["No production/native lifecycle/host resume or unattended activation claim",
                      "No model chosen, no persisted permission/trust or central Store modification requested"]}
(root / "summary.json").write_text(json.dumps(summary, indent=2) + "\n")
print(json.dumps({"finished": True, "passed": sum(x["transport_pass"] for x in results),
                  "repository_content_drift": drift}), flush=True)
sys.exit(0 if all(x["transport_pass"] for x in results) and not drift else 1)
