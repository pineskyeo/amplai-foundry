import importlib.util
import json
import pathlib
import shutil
import subprocess
import sys

ROOT = pathlib.Path('/tmp/w003-r5-recovery.qESisa/checkout').resolve()
sys.path[:0] = [str(ROOT / 'scripts'), str(ROOT / 'tests'), str(ROOT / 'src')]
from ai import test_document_lifecycle as docs
from ai import test_document_typed_visibility as typed
from ai import test_document_html as views
import amplai_docs as engine

assert pathlib.Path(engine.__file__).resolve() == ROOT / 'scripts/amplai_docs.py'
BASE = pathlib.Path('/tmp/w003-r5-recovery.qESisa/optional-alias-delivery').resolve()
assert not BASE.exists()
BASE.mkdir()
root, release_path = views.fixture(engine, BASE, security='PUBLIC')
(root / 'assets').mkdir()
outside = BASE / 'outside-synthetic.bin'
outside.write_bytes(b'SYNTHETIC_NONSECRET_FIXTURE\n')
(root / 'assets/PRIVATE_CANARY.bin').symlink_to(outside)
typed.policy_routes(root, [typed.route('assets', 'RESTRICTED')], hint='known_absent')
docs.document(root, security='PUBLIC', body='# Guide\nSee `assets/PRIVATE_CANARY.bin`.\n')
docs.reviewed(engine, root)
record = engine.read_document(root, 'docs/guide.md')
store = json.loads((root / '.ai-team/knowledge/document-reviews.json').read_text())
review = next(row for row in store['reviews'] if row['doc_id'] == 'canary:guide')
release = json.loads((root / release_path).read_text())
release['documents'][0] = {'doc_id': 'canary:guide', 'source_sha256': record['source_sha256'], 'snapshot_sha256': engine.object_digest(record['snapshot']), 'review_sha256': engine.object_digest(review)}
verification = release.pop('verification')
proof = json.loads((root / verification['path']).read_text())
proof['input_sha256'] = engine.object_digest(release)
docs.put_json(root / verification['path'], proof)
verification['sha256'] = views.sha((root / verification['path']).read_bytes())
release['verification'] = verification
docs.put_json(root / release_path, release)
try:
    value = engine.prepare_view_input(root, release_path, audience='operator', security='PUBLIC')
except engine.DocumentError as exc:
    html_result = {'code': exc.code, 'marker_emitted': False}
else:
    html_result = {'code': 'SUCCESS', 'marker_emitted': 'PRIVATE_CANARY' in json.dumps(value)}

installed = BASE / 'installed'
installed.mkdir()
install_command = [sys.executable, '-B', str(ROOT / 'tools/amplai-loop-kit/install.py'), '--target', str(installed), '--bootstrap-baseline', '--app-id', 'recovery-canary', '--project-id', 'canary', '--no-git']
installation = subprocess.run(install_command, capture_output=True, text=True, timeout=30)
assert installation.returncode == 0, installation.stderr
assert (installed / 'scripts/amplai_docs.py').read_bytes() == (ROOT / 'scripts/amplai_docs.py').read_bytes()
for directory in ('docs', 'src', 'assets', 'evidence'):
    shutil.copytree(root / directory, installed / directory, symlinks=True, dirs_exist_ok=True)
for name in ('.ai-team/policy/documentation.json', '.ai-team/knowledge/document-reviews.json'):
    destination = installed / name
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(root / name, destination)
spec = importlib.util.spec_from_file_location('installed_alias_probe', installed / 'scripts/amplai_docs.py')
installed_engine = importlib.util.module_from_spec(spec)
exec(compile((installed / 'scripts/amplai_docs.py').read_bytes(), str(installed / 'scripts/amplai_docs.py'), 'exec'), installed_engine.__dict__)
docs.reviewed(installed_engine, installed)
commands = []
for command in ('inventory', 'query', 'index'):
    for history in (False, True):
        args = ['docs', command, '--release', 'release-2', '--security', 'PUBLIC']
        if history:
            args.append('--history')
        actual = docs.loop_command(installed, *args)
        commands.append({'command': 'python -I -S -B scripts/loopctl.py ' + ' '.join(args), 'exit_code': actual.returncode, 'marker_emitted': 'PRIVATE_CANARY' in actual.stdout + actual.stderr})

docs.document(root, security='PUBLIC', body='# Guide\nRequires `assets/PRIVATE_CANARY.bin`.\n')
try:
    engine.read_document(root, 'docs/guide.md')
except engine.DocumentError as exc:
    explicit_control = {'code': exc.code, 'marker_emitted': 'PRIVATE_CANARY' in str(exc)}
else:
    explicit_control = {'code': 'SUCCESS'}
print(json.dumps({'install_exit': installation.returncode, 'installed_source_byte_equal': True, 'public_view_input': html_result, 'installed_cli': commands, 'explicit_control': explicit_control, 'fixture': str(BASE)}, indent=2))
