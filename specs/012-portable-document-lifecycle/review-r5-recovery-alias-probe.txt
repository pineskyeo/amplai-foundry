import importlib.util
import json
import pathlib
import sys
import tempfile

ROOT = pathlib.Path('/tmp/w003-r5-recovery.qESisa/checkout').resolve()
sys.path[:0] = [str(ROOT / 'scripts'), str(ROOT / 'tests'), str(ROOT / 'src')]
from ai import test_document_lifecycle as docs
from ai import test_document_typed_visibility as typed

assert docs.ROOT == ROOT
results = []
for engine_path in ('scripts/amplai_docs.py', 'tools/amplai-loop-kit/baseline/payload/scripts/amplai_docs.py'):
    source = ROOT / engine_path
    spec = importlib.util.spec_from_file_location('optional_alias_probe', source)
    engine = importlib.util.module_from_spec(spec)
    exec(compile(source.read_bytes(), str(source), 'exec'), engine.__dict__)
    assert pathlib.Path(engine.__file__).resolve().is_relative_to(ROOT)
    for outside in (False, True):
        with tempfile.TemporaryDirectory(prefix='optional-alias-', dir='/tmp/w003-r5-recovery.qESisa') as d:
            base = pathlib.Path(d)
            root = docs.repository(base)
            assets = root / 'assets'
            assets.mkdir()
            target = (base if outside else assets) / 'target.bin'
            target.write_bytes(b'SYNTHETIC_TEST_ONLY\n')
            (assets / 'PRIVATE_CANARY.bin').symlink_to(target)
            typed.policy_routes(root, [typed.route('assets', 'RESTRICTED')], hint='known_absent')
            docs.document(root, security='PUBLIC', body='# Guide\nSee `assets/PRIVATE_CANARY.bin`.\n')
            docs.reviewed(engine, root)
            record = engine.read_document(root, 'docs/guide.md')
            for consumer in engine.CURRENT_CONSUMERS:
                for history in (False, True):
                    try:
                        result = engine.select_documents(root, 'release-2', security='PUBLIC', consumer=consumer, history=history)
                    except engine.DocumentError as exc:
                        outcome = {'code': exc.code, 'marker_emitted': False}
                    else:
                        outcome = {'code': 'SUCCESS', 'marker_emitted': 'PRIVATE_CANARY' in json.dumps(result)}
                    results.append({'engine': engine_path, 'outside_symlink': outside, 'consumer': consumer, 'history': history, 'freshness': record['freshness'], **outcome})
print(json.dumps(results, indent=2))
