"""Bounded in-memory mutation controls against the frozen R5 regression tests."""
import inspect
import json
from pathlib import Path
from types import SimpleNamespace

import pytest

from ai import test_document_reference_resources as resources
from ai import test_document_typed_visibility as typed

ROOT = Path('/tmp/w003-r5-regression.cJS4Cl/repo').resolve()
OUT = Path('/tmp/w003-r5-regression.cJS4Cl')
assert Path(resources.__file__).resolve().is_relative_to(ROOT)
assert Path(typed.__file__).resolve().is_relative_to(ROOT)
results = []


def run_mutant(name, invoke):
    try:
        invoke()
    except (AssertionError, pytest.fail.Exception) as exc:
        results.append({'mutation': name, 'status': 'KILLED',
                        'assertion': str(exc), 'exception': type(exc).__name__})
    else:
        results.append({'mutation': name, 'status': 'SURVIVED'})


def replace_function(engine, name, old, new):
    source = inspect.getsource(getattr(engine, name))
    assert source.count(old) == 1
    exec(compile(source.replace(old, new), engine.__file__, 'exec', dont_inherit=True),
         engine.__dict__)


for index, source in enumerate([
    'scripts/amplai_docs.py',
    'tools/amplai-loop-kit/baseline/payload/scripts/amplai_docs.py',
]):
    engine = typed.typed_engine.__wrapped__(SimpleNamespace(param=source))
    assert Path(engine.__file__).resolve() == ROOT / source
    replace_function(engine, 'reference_visibility',
                     'for raw in extract_typed_references(text):', 'for raw in []:')
    case = OUT / ('mutant-typed-' + str(index))
    case.mkdir()
    run_mutant('remove_typed_visibility_' + str(index), lambda: (
        typed.test_typed_security_precedes_all_consumers_and_history(
            engine, case, 'inline', 'RESTRICTED')))

    engine = resources.bounded_engine.__wrapped__(SimpleNamespace(param=source))
    replace_function(engine, 'classify_document_reference',
                     'budget.spend(1 + sum(len(item.get(key) or "") for key in fields))',
                     'pass  # removed aggregate classification precharge')
    with pytest.MonkeyPatch.context() as patch:
        run_mutant('remove_classification_precharge_' + str(index), lambda: (
            resources.test_s16_classification_charges_before_prefix_processing(engine, patch)))


def old_cache_loader(request):
    import importlib.util
    path = resources.docs.ROOT / request.param
    spec = importlib.util.spec_from_file_location('resource_engine_mutant', path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


case = OUT / 'mutant-cache-loader'
case.mkdir()
with pytest.MonkeyPatch.context() as patch:
    patch.setattr(resources, 'bounded_engine', SimpleNamespace(__wrapped__=old_cache_loader))
    run_mutant('restore_cache_writing_loader', lambda: (
        resources.test_reference_fixture_import_never_writes_package_bytecode(case, patch)))

print(json.dumps({'results': results, 'killed': sum(r['status'] == 'KILLED' for r in results),
                  'total': len(results), 'mutation_scope': 'in-memory only; synthetic cache outside repo'}, indent=2))
assert all(r['status'] == 'KILLED' for r in results)
